{{- /* Renders an author list. ctx: dict "authors" slice, "max" int (0 = all), "link" bool. Owner in bold. */ -}} {{- $people := site.Data.people -}} {{- $max := .max | default 0 -}} {{- $link := .link -}} {{- $n := len .authors -}} {{- $out := slice -}} {{- range $i, $a := .authors -}} {{- $p := index $people $a -}} {{- $label := cond (and (ne $p nil) (not $.full)) (index $p "short") $a -}} {{- $html := "" -}} {{- if and $p (index $p "owner") -}} {{- $html = printf `%s` $label -}} {{- else if and $p $link (index $p "slug") -}} {{- $html = printf `%s` (printf "authors/%s/" (index $p "slug") | relURL) $label -}} {{- else -}} {{- $html = $label -}} {{- end -}} {{- $out = $out | append $html -}} {{- end -}} {{- if and (gt $max 0) (gt $n (add $max 1)) -}} {{- $owner_idx := -1 -}} {{- range $i, $a := .authors }}{{ if eq $a site.Params.owner.name }}{{ $owner_idx = $i }}{{ end }}{{ end -}} {{- $first := first $max $out -}} {{- $rest := after $max $out -}} {{ delimit $first ", " | safeHTML }}, {{ delimit $rest ", " | safeHTML }} {{- else -}} {{ delimit $out ", " | safeHTML }} {{- end -}}
